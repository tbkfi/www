# ADDR, ID
lspci -nn | grep -i "e223\|Arc Pro B70"


# Set VF count to 0 (match the address to main GPU address!)
echo 0 > /sys/bus/pci/devices/0000\:03\:00.0/sriov_numvfs
# Unbind current driver
root@pve:~# echo "0000:03:00.1" > /sys/bus/pci/devices/0000\:03\:00.1/driver/unbind 
# Inform vfio-pci of new device
root@pve:~# echo "8086 e223" > /sys/bus/pci/drivers/vfio-pci/new_id 
# Bind new device to vfio-pci
echo 0000:03:00.1 > /sys/bus/pci/drivers/vfio-pci/bind

# One liner: Find GPU set PWR LIM to 180W
echo 180000000 > $(find /sys/devices -path "*/0000:03:00.0/hwmon/hwmon*/power1_cap" 2>/dev/null | head -1)
